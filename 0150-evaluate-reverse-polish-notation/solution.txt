class Solution {
public:
    string minRemoveToMakeValid(string s) {
        int bal=0;
        string res="";
        string ans="";
        for(char c:s){
            if(c=='('){
                bal++;
                res+=c;
            }else if(c==')'){
                if(bal>0){
                    bal--;
                    res+=c;
                }
            }else{
                res+=c;
            }
        }
        for(int i=res.length()-1;i>=0;i--){
            if(res[i]=='('&& bal>0){
                bal--;
                continue;
            }
            ans+=res[i];
        }
        reverse(ans.begin(),ans.end());
        return ans;
    }
};
