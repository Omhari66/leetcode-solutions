class Solution {
public:
    int largestAltitude(vector<int>& gain) {
        int maxAlt=0;
        int n=gain.size();
        // vector<int>prefix(n+1,0);
        int curAlt=0;
        for(int i=0;i<n;i++){
            curAlt+=gain[i];
            maxAlt=max(curAlt,maxAlt);
        }
        return maxAlt;
    }
};
