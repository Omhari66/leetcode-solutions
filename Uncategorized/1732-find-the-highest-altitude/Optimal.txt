class Solution {
public:
    int largestAltitude(vector<int>& gain) {
        int maxAlt=0;
        int n=gain.size();
        vector<int>prefix(n+1,0);
        prefix[0]=0;
        for(int i=0;i<n;i++){
            prefix[i+1]=prefix[i]+gain[i];
            maxAlt=max(prefix[i+1],maxAlt);
        }
        return maxAlt;
    }
};
