class Solution {
private:
int dfs(int r,int c,vector<vector<int>>&grid){
    int rows=grid.size();
    int cols=grid[0].size();

    if(r<0||c<0||r>=rows||c>=cols||grid[r][c]==0){
        return 0;
    }
    
    grid[r][c]=0;
    int area=1+dfs(r,c-1,grid)+dfs(r,c+1,grid)+dfs(r-1,c,grid)+dfs(r+1,c,grid);

    return area;
}
public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int n=grid.size();
        int m=grid[0].size();
        int maxArea=0;
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(grid[i][j]==1){
                   maxArea= max(maxArea,dfs(i,j,grid));
                }
            }
        }
            return maxArea;
    }
};
