class Solution {
public:
    int addMinimum(string word) {
       string exp="abc";
       int j=0;
       int i=0;
       int ans=0;
       while(i<word.size()){
        if(exp[j%3]==word[i]){
            i++;
            j++;
        }
        else{
            j++;
            ans++;
        }
       }
       ans+=(3-j%3)%3;
       return ans;
    }
};
