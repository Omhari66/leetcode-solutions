class Solution {
public:
    string predictPartyVictory(string senate) {
     int n=senate.length();
     queue<int>r;
     queue<int>d;
     for(int i=0;i<n;i++){
        if(senate[i]=='R'){
            r.push(i);
        }else{
            d.push(i);
        }
     }   
     while(!r.empty()&&!d.empty()){
        int ri=r.front();
        int di=d.front();
        if(ri>di){
            r.pop();
            d.pop();
            d.push(di+n);
        }else{
            d.pop();
            r.pop();
            r.push(ri+n);
        }
     }
     return r.empty()?"Dire":"Radiant";
    }
};
